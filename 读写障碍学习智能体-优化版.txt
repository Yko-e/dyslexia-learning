function switchTab(name) {
    document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
    document.querySelector(`.tab-btn[data-tab="${name}"]`).classList.add('active');
    document.querySelectorAll('.tab-page').forEach(p => p.classList.remove('active'));
    document.getElementById('tab-' + name).classList.add('active');
    window.scrollTo(0, 0);
    if (name === 'ai') renderAIWelcome();
}

function renderLessonPicker() {
    const picker = document.getElementById('lessonPicker');
    picker.innerHTML = '';
    // 学段切换按钮
    Object.keys(textbook).forEach(stage => {
        const btn = document.createElement('button');
        btn.className = 'lesson-btn' + (stage === currentStage ? ' active' : '');
        btn.innerText = stage;
        btn.onclick = () => { currentStage = stage; renderLessonPicker(); loadFirstLesson(); };
        picker.appendChild(btn);
    });
    // 分隔线
    const sep = document.createElement('div');
    sep.style.cssText = 'width:100%; border-top:1px dashed rgba(130,130,130,0.4); margin:8px 0;';
    picker.appendChild(sep);
    // 内置课文
    textbook[currentStage].forEach((art, i) => {
        const b = document.createElement('button');
        b.className = 'lesson-btn' + (currentArticle && currentArticle.title === art.title && !currentArticle.isCustom ? ' active' : '');
        b.innerText = art.title;
        b.onclick = () => openLesson(i, false);
        picker.appendChild(b);
    });
    // 用户自定义课文（带⭐标记，可删除：桌面右键 / 手机长按）
    const customs = customList[currentStage] || [];
    customs.forEach((art, i) => {
        const b = document.createElement('button');

        b.innerText = '⭐ ' + art.title;

        // iOS：长按 600ms 删除（长按时震动提示）

                if (navigator.vibrate) navigator.vibrate(10);
                if (confirm('删除这篇自定义课文「' + art.title + '」？')) deleteCustomLesson(i);
            }, 600);
        }, { passive: true });
        b.addEventListener('touchend', function() { clearTimeout(pressTimer); });
        b.addEventListener('touchmove', function() { clearTimeout(pressTimer); });
        picker.appendChild(b);
    });
    // 提示
    if (customs.length > 0) {
        const tip = document.createElement('div');
        tip.style.cssText = 'width:100%; font-size:12px; color:#887766; margin-top:4px;';
        tip.innerText = '⭐ 为自定义课文，右键（长按）可删除';

// 添加自定义课文：自动分句 + 自动生成基础题目
function addCustomLesson() {
    const stage = document.getElementById('addStage').value;
    const title = document.getElementById('addTitle').value.trim();
    const contentRaw = document.getElementById('addContent').value.trim();
    const status = document.getElementById('addLessonStatus');

    if (!title || !contentRaw) {
        status.innerText = '⚠️ 请填写标题和正文';
        return;
    }
    if (title.length > 30) {
        status.innerText = '⚠️ 标题请控制在30字以内';
        return;
    }

    // 自动分句：按标点和换行拆分
    let sentences = contentRaw
        .split(/\n+/)
        .map(line => line.trim())
        .filter(Boolean)
        .flatMap(line => line.split(/(?<=[。！？；])\s*/).filter(s => s.trim()))
        .map(s => s.trim());

    if (sentences.length === 0) {
        status.innerText = '⚠️ 正文内容为空，请检查';
        return;
    }
    if (sentences.length > 60) {
        sentences = sentences.slice(0, 60);
        status.innerText = '⚠️ 正文过长，已截取前60句';
    }

    // 自动生成基础题目（从正文中挖词填空 + 内容理解）
    const quiz = [];
    sentences.forEach((sentence, i) => {
        if (quiz.length >= 3) return;
        const chars = sentence.replace(/[，。！？；、“”\s]/g, '').split('');
        if (chars.length >= 4 && chars.length <= 12) {
            const pos = Math.floor(chars.length / 2);
            const answer = chars[pos];
            const qText = sentence.replace(new RegExp('(' + answer + ')'), '____');
            quiz.push({ type: 'fill', q: qText + '（第' + (i + 1) + '句挖空）', answer: answer });
        }
    });
    quiz.push({
        type: 'choice',
        q: '《' + title + '》这篇课文主要写了什么？',
        options: ['景物或事情的内容', '一种道理', '一个人物'],
        answer: 0
    });

    customList[stage].push({ title: title, content: sentences, quiz: quiz });
    saveCustomLessons();
    loadCustomLessons();
    renderLessonPicker();
    // 打开刚添加的课文
    openLesson(customList[stage].length - 1, true);

    status.innerText = '✅ 已保存「' + title + '」，共' + sentences.length + '句、' + quiz.length + '道题';
    document.getElementById('addTitle').value = '';
    document.getElementById('addContent').value = '';
}

    saveCustomLessons();
    loadCustomLessons();
    renderLessonPicker();
    loadFirstLesson();
}

function renderText() {
    const wrap = document.getElementById('textContainer');
    wrap.innerHTML = '';
    // 根据阅读难度决定是否拆分短句
    const sentences = splitForRead(currentArticle.content);
    sentences.forEach(s => {
        const div = document.createElement('div');
        div.className = 'sentence-item';
        div.innerText = s;
        div.onclick = () => readSentence(s, div);
        wrap.appendChild(div);
    });
    applyReadMode();
}

// 阅读难度处理：拆短句
function splitForRead(lines) {
    if (readLevel === 'light') {
        // 轻度：保留原文句子，不做额外拆分
        return lines;
    }
    // 中度/重度：把长句按标点拆成更短的分句
    const result = [];
    lines.forEach(line => {
        const parts = line.split(/(?<=[，。；！？、])/).filter(s => s.trim());
        if (parts.length === 0) result.push(line);
        else parts.forEach(p => result.push(p.trim()));
    });
    return result;
}

    } else {

    }
}

// ========== 朗读功能（iOS 兼容） ==========
// iOS Safari 需要 webkit 前缀检测 + 中文语音选择 + 分段朗读防截断
function speechSupported() {
    return ('speechSynthesis' in window) || ('webkitSpeechSynthesis' in window);
}
function getZhVoice() {
    const synth = window.speechSynthesis || window.webkitSpeechSynthesis;
    if (!synth) return null;
    const voices = synth.getVoices();
    // iOS 上语音列表是异步加载的，首次可能为空
    if (voices.length === 0) {
        synth.onvoiceschanged = function() {
            const v = synth.getVoices().find(v => v.lang && v.lang.toLowerCase().indexOf('zh') === 0);
            if (v) speech.voice = v;
        };
        return null;
    }
    return voices.find(v => v.lang && v.lang.toLowerCase().indexOf('zh') === 0) || null;
}

function readSentence(text, dom) {
    if (!speechSupported()) { alert('当前浏览器不支持语音朗读'); return; }
    const synth = window.speechSynthesis || window.webkitSpeechSynthesis;
    synth.cancel();
    const u = new SpeechSynthesisUtterance(text);
    u.lang = 'zh-CN';
    u.rate = speech.rate;
    const v = getZhVoice(); if (v) u.voice = v;
    document.querySelectorAll('.sentence-item').forEach(s => s.classList.remove('active-read'));
    if (dom) dom.classList.add('active-read');
    synth.speak(u);
}

// 分段朗读全文：iOS 会把长文本截断，这里按句逐段朗读
function readAll() {
    if (!speechSupported()) { alert('当前浏览器不支持语音朗读'); return; }
    const synth = window.speechSynthesis || window.webkitSpeechSynthesis;
    synth.cancel();
    const sentences = splitForRead(currentArticle.content);
    readQueue = sentences.slice();
    speakNextSegment();
}

function speakNextSegment() {
    const synth = window.speechSynthesis || window.webkitSpeechSynthesis;
    if (!synth || readQueue.length === 0) return;
    const seg = readQueue.shift();
    const u = new SpeechSynthesisUtterance(seg);
    u.lang = 'zh-CN';
    u.rate = speech.rate;
    const v = getZhVoice(); if (v) u.voice = v;
    u.onend = function() { speakNextSegment(); };
    u.onerror = function() { /* iOS 上可能报 interrupted，忽略继续 */ };
    synth.speak(u);
}

function randomQuiz() {
    const total = currentArticle.quiz.length;
    if (usedQuizIndexes.length >= total) usedQuizIndexes = [];
    let randomIdx;
    do { randomIdx = Math.floor(Math.random() * total); } while (usedQuizIndexes.includes(randomIdx));
    usedQuizIndexes.push(randomIdx);
    currentQuizIndex = randomIdx;
    orderSelected = [];
    renderQuiz();
}

function renderQuiz() {
    const quiz = currentArticle.quiz[currentQuizIndex];
    const box = document.getElementById('quizBox');
    const typeDom = document.getElementById('quizType');
    document.getElementById('quizFeedback').innerText = '';

    const typeMap = { choice:'📝 选择题', fill:'✏️ 填空题', order:'🔢 排序题' };
    typeDom.innerText = typeMap[quiz.type] || '练习题';

    if (quiz.type === 'choice') {
        box.innerHTML = `<div style="margin-bottom:10px;">${quiz.q}</div>`;
        // 简单难度：3选项+答错提示；中等/困难：全选项
        let opts = quiz.options;
        const prefix = quizLevel === 'easy' ? '选项较少，慢慢选：' : '请选择答案：';
        box.innerHTML = `<div style="font-size:14px;color:#887766;margin-bottom:4px;">${prefix}</div>${box.innerHTML}`;
        opts.forEach((opt, i) => {
            box.innerHTML += `<button class="option-btn" onclick="checkChoice(${i})">${String.fromCharCode(65+i)}. ${opt}</button>`;
        });
    } else if (quiz.type === 'fill') {
        // 困难难度：无提示填空；简单/中等：给出首字提示
        const hint = quizLevel === 'hard' ? '' : `（提示：答案第一个字是「${quiz.answer.charAt(0)}」）`;
        box.innerHTML = `
            <div style="margin-bottom:10px;">${quiz.q.replace('____', '<input type="text" class="fill-blank" id="fillInput" placeholder="填写答案">')}</div>
            <div style="font-size:14px;color:#887766;margin-bottom:6px;">${hint}</div>
            <button onclick="checkFill()">提交答案</button>
        `;
    }
}

function checkChoice(selectIdx) {
    const quiz = currentArticle.quiz[currentQuizIndex];
    const feedback = document.getElementById('quizFeedback');
    currentTotal++;
    if (selectIdx === quiz.answer) {
        currentCorrect++;
        feedback.innerText = '🎉 答对啦！真棒！';
        feedback.style.color = '#228822';
    } else {
        feedback.innerText = '😊 没关系，正确答案是：' + quiz.options[quiz.answer];
        feedback.style.color = '#bb4444';
        // 简单难度：答错给额外提示
        if (quizLevel === 'easy') {
            feedback.innerHTML += '<div class="hint-box">💡 提示：回到课文里找找答案哦，把相关句子再读一遍。</div>';
        }
    }
    updateAccuracy();
    saveStat();
}

function checkFill() {
    const quiz = currentArticle.quiz[currentQuizIndex];
    const input = document.getElementById('fillInput').value.trim();
    const feedback = document.getElementById('quizFeedback');
    currentTotal++;
    if (input === quiz.answer) {
        currentCorrect++;
        feedback.innerText = '🎉 答对啦！';
        feedback.style.color = '#228822';
    } else {
        feedback.innerText = '😊 差一点点，正确答案是：' + quiz.answer;
        feedback.style.color = '#bb4444';
        if (quizLevel !== 'hard') {
            feedback.innerHTML += '<div class="hint-box">💡 提示：把课文里含有这个空的那句话读一遍。</div>';
        }
    }
    updateAccuracy();
    saveStat();
}

function nextQuestion() { randomQuiz(); }

function updateAccuracy() {
    const rate = currentTotal > 0 ? Math.round(currentCorrect / currentTotal * 100) : 0;
    document.getElementById('currentAcc').innerText = rate + '%';
}

// 测试连接
async function testAIConfig() {
    const cfg = getAIConfig();
    const status = document.getElementById('aiConfigStatus');
    if (!cfg.baseUrl || !cfg.apiKey || !cfg.model) {
        status.innerText = '⚠️ 请先填写完整的接口地址、Key 和模型名';
        return;
    }
    status.innerText = '⏳ 正在测试连接...';
    try {
        const res = await fetch(cfg.baseUrl.replace(/\/$/, '') + '/chat/completions', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': 'Bearer ' + cfg.apiKey
            },
            body: JSON.stringify({
                model: cfg.model,
                messages: [{ role: 'user', content: '你好，请回复：连接成功' }],
                max_tokens: 20
            })
        });
        if (!res.ok) {
            const errText = await res.text();
            status.innerText = '❌ 连接失败：HTTP ' + res.status + '（' + errText.slice(0, 80) + '）。\n若提示跨域(CORS)问题，请使用支持浏览器跨域的 API 网关或本地代理转发。';
            return;
        }
        const data = await res.json();
        status.innerText = '✅ 连接成功！模型回复：' + (data.choices && data.choices[0] ? data.choices[0].message.content.slice(0, 30) : '正常');
    } catch (e) {
        status.innerText = '❌ 连接失败：' + e.message + '。常见原因：跨域限制(CORS)、接口地址错误。建议使用支持 CORS 的 API 平台或本地代理。';
    }
}

// 发送 AI 提问

    const input = document.getElementById('aiInput');

function askAI(question) {
    const box = document.getElementById('aiChatBox');
    // 显示用户问题
    box.innerHTML += `<div class="chat-msg user">${escapeHtml(question)}</div>`;
    const loading = document.createElement('div');
    loading.className = 'chat-msg ai ai-loading';
    loading.innerText = '思考中...';
    box.appendChild(loading);
    box.scrollTop = box.scrollHeight;

    const cfg = getAIConfig();
    const systemPrompt = buildSystemPrompt();
    const context = '当前课文是《' + currentArticle.title + '》。\n课文内容：\n' + currentArticle.content.join('\n') + '\n';

    if (cfg.baseUrl && cfg.apiKey && cfg.model) {
        // 有配置：调用真实 API
        fetch(cfg.baseUrl.replace(/\/$/, '') + '/chat/completions', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': 'Bearer ' + cfg.apiKey
            },
            body: JSON.stringify({
                model: cfg.model,
                messages: [
                    { role: 'system', content: systemPrompt },
                    { role: 'user', content: context + '\n学生的问题：' + question }
                ],
                temperature: 0.7,
                max_tokens: 600
            })
        })
        .then(res => {
            if (!res.ok) throw new Error('HTTP ' + res.status);
            return res.json();
        })
        .then(data => {
            loading.remove();
            const answer = data.choices && data.choices[0] ? data.choices[0].message.content : '（无返回内容）';
            box.innerHTML += `<div class="chat-msg ai">${escapeHtml(answer)}</div>`;
            box.scrollTop = box.scrollHeight;
        })
        .catch(err => {
            loading.remove();
            box.innerHTML += `<div class="chat-msg ai">⚠️ 调用失败：${escapeHtml(err.message)}。请检查「设置」里的 AI 配置；若为跨域(CORS)限制，请使用支持 CORS 的接口或本地代理。现在改用离线答疑：<br>${offlineAnswer(question)}</div>`;
            box.scrollTop = box.scrollHeight;
        });
    } else {
        // 未配置：离线答疑降级
        loading.remove();
        box.innerHTML += `<div class="chat-msg ai">${offlineAnswer(question)}<br><br><span style="font-size:13px;color:#887766;">（提示：你还没有配置 AI 接口，当前是内置离线答疑。到「设置」填入你的 API 后可获得更完整的回答。）</span></div>`;
        box.scrollTop = box.scrollHeight;
    }
}

// 离线答疑（基于本地词典与课文规则）
function offlineAnswer(question) {
    // 1. 词义查询
    for (const word in wordDict) {
        if (question.includes(word)) {
            const d = wordDict[word];
            return `「${word}」${d.pinyin}：${d.explain}。\n例句：${d.example}`;
        }
    }
    // 2. 课文相关
    if (question.includes('讲什么') || question.includes('主要内容')) {
        return `这篇《${currentArticle.title}》我们这样理解：\n` + currentArticle.content.slice(0, 3).map((s, i) => `第${i+1}句：${s}`).join('\n') + `\n\n先读熟每一句，再连起来理解，会容易很多哦。`;
    }
    if (question.includes('字词') || question.includes('难')) {
        return `课文里可以重点认这些字词：${Object.keys(wordDict).join('、')}。\n选中课文里的字词，会自动弹出解释哦。`;
    }
    // 3. 兜底
    return '这个问题我需要想一想。建议你先在「设置」里配置 AI 接口（支持 OpenAI 兼容 API），我就能用大模型详细解答你的问题啦。现在可以先选中课文里的字词查看释义，或点击上方快捷按钮提问。';
}

// 构建 AI 系统提示词（人格 + 难度适配）

    const quizDesc = { easy:'学生答题水平简单，讲解要非常浅显。', middle:'学生答题水平中等，可适当展开。', hard:'学生答题水平较好，可以讲解得完整一些。' };
    return `你是「读写障碍学习智能体」的AI答疑老师，服务对象是有阅读困难的中小学生。
规则：
1. 语气温和、耐心、鼓励，绝不说批评的话。
2. 回答要用简单、短小的句子，根据学生困难程度调整：${readDesc[readLevel]}
3. 答题难度参考：${quizDesc[quizLevel]}
4. 结合学生正在学的课文内容来回答。
5. 你只提供学习辅助，不做任何医学诊断。`;
}

}

    rulerOpen = !rulerOpen;
    const rulerDom = document.getElementById('ruler');
    if (rulerOpen) {
        rulerDom.classList.add('show');
        document.addEventListener('mousemove', moveRuler);
        document.addEventListener('touchmove', touchMoveRuler, { passive: true });
    } else {
        rulerDom.classList.remove('show');
        document.removeEventListener('mousemove', moveRuler);
        document.removeEventListener('touchmove', touchMoveRuler);
    }
}
function moveRuler(e) { document.getElementById('ruler').style.top = (e.clientY - 20) + 'px'; }
function touchMoveRuler(e) { if (e.touches[0]) document.getElementById('ruler').style.top = (e.touches[0].clientY - 20) + 'px'; }

function toggleSimple() { document.body.classList.toggle('simple-mode'); }

// 十一、学习记录（localStorage）
// ============================================================
function saveStat() {
    const data = JSON.parse(localStorage.getItem('readAgentStat') || '{}');
    data.correct = (data.correct || 0);
    data.total = (data.total || 0) + 1;
    localStorage.setItem('readAgentStat', JSON.stringify(data));
    refreshRecordStats();
}
function refreshRecordStats() {
    const data = JSON.parse(localStorage.getItem('readAgentStat') || '{}');
    const total = data.total || 0;
    const correct = data.correct || 0;
    const rate = total > 0 ? Math.round(correct / total * 100) : 0;
    document.getElementById('statTotal').innerText = total + ' 道';
    document.getElementById('statCorrect').innerText = correct + ' 道';
    document.getElementById('statRate').innerText = rate + '%';
    document.getElementById('statReadLevel').innerText = { light:'轻度', middle:'中度', heavy:'重度' }[readLevel];
    document.getElementById('statQuizLevel').innerText = { easy:'简单', middle:'中等', hard:'困难' }[quizLevel];
}
function resetStudy() {
    localStorage.removeItem('readAgentStat');
    refreshRecordStats();
}
